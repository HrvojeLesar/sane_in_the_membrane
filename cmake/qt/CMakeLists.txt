cmake_minimum_required(VERSION 4.2)
project(qt_builder)

include(ExternalProject)

if(QT6_INSTALL_PREFIX STREQUAL "")
    set(QT6_INSTALL_PREFIX "${CMAKE_BINARY_DIR}/qt6-install" CACHE PATH "Qt6 install prefix" FORCE)
endif()

set(QT6_BUILD_SUBMODULES "qtsvg\\\;qtbase")

set(QT6_GIT_SUBMODULES "")
string(REPLACE "\\\;" " " QT6_GIT_SUBMODULES ${QT6_BUILD_SUBMODULES})

set(QT6_CMAKE_ARGS
    -DCMAKE_BUILD_TYPE=${CMAKE_BUILD_TYPE}
    -DCMAKE_INSTALL_PREFIX=<INSTALL_DIR>
    -DBUILD_SHARED_LIBS=OFF
    -DQT_BUILD_TESTS=OFF
    -DQT_BUILD_EXAMPLES=OFF
    -DQT_FEATURE_static=ON
    -DQT_FEATURE_windows=ON
    -DQT_BUILD_SUBMODULES=${QT6_BUILD_SUBMODULES}
)

ExternalProject_Add(qt6
    GIT_REPOSITORY https://code.qt.io/qt/qt5.git
    GIT_TAG v6.10.2
    GIT_SHALLOW TRUE
    GIT_SUBMODULES ${QT6_GIT_SUBMODULES}
    GIT_SUBMODULES_RECURSE TRUE
    SOURCE_DIR ${CMAKE_BINARY_DIR}/_deps/qt6-src
    BINARY_DIR ${CMAKE_BINARY_DIR}/_deps/qt6-build
    INSTALL_DIR ${QT6_INSTALL_PREFIX}
    CMAKE_ARGS ${QT6_CMAKE_ARGS}

    BUILD_COMMAND ${CMAKE_COMMAND} --build <BINARY_DIR> --parallel
    INSTALL_COMMAND ${CMAKE_COMMAND} --install <BINARY_DIR>
)
